<script>
// 1. DATABASE OF REAL POPULAR GAMES
const realGames = [
{ title: "1v1.LOL", category: "Action", img: "https://1v1.lol" },
{ title: "Shell Shockers", category: "Action", img: "https://shellshock.io" },
{ title: "Slope", category: "Skill", img: "https://slopegame.io" },
{ title: "Retro Bowl", category: "Sports", img: "https://via.placeholder.com" },
{ title: "Drive Mad", category: "Driving", img: "https://img.gamedistribution.com" },
{ title: "Tunnel Rush", category: "Skill", img: "https://edupure.github.io" },
{ title: "Minecraft Classic", category: "Classic", img: "https://classic.minecraft.net" },
{ title: "Basketball Stars", category: "Sports", img: "https://edupure.github.io" },
{ title: "Geometry Dash", category: "Skill", img: "https://upload.wikimedia.org" },
{ title: "Subway Surfers", category: "Skill", img: "https://via.placeholder.com" }
];

// 2. GENERATOR ENGINE (REACHES 1000)
const adjs = ["Cyber", "Neon", "Super", "Hyper", "Dark", "Epic", "Mega", "Ultimate", "Crazy", "Alpha"];
const nouns = ["Drift", "Striker", "Combat", "Runner", "Quest", "Jump", "Dash", "Battle", "Baller", "Slayer"];
const cats = ["Action", "Skill", "Sports", "Driving"];

const games = [...realGames];

for (let i = games.length; i < 1000; i++) {
const title = adjs[i % 10] + " " + nouns[Math.floor(Math.random() * 10)] + " " + (i + 1);
games.push({
title: title,
category: cats[i % 4],
img: `https://picsum.photos{i + 123}/200/120` // Unique high-quality visuals
});
}

// 3. RENDER ENGINE
function display() {
const search = document.getElementById('searchBar').value.toLowerCase();
const cat = document.getElementById('filterOptions').value;
const grid = document.getElementById('gameGrid');

const filtered = games.filter(g => {
const mSearch = g.title.toLowerCase().includes(search);
const mCat = cat === "All" || g.category === cat;
return mSearch && mCat;
});

// Limit initial render to 100 for speed, show more on search
grid.innerHTML = filtered.slice(0, 500).map(g => `
<a href="#" class="game-card" onclick="alert('Starting ${g.title}...')">
<img src="${g.img}" loading="lazy">
<div class="game-info">
<h3 class="game-title">${g.title}</h3>
<span class="game-cat">${g.category}</span>
</div>
</a>
`).join('');

document.getElementById('zoneCount').innerText = `${filtered.length} Games Unblocked`;
}

document.getElementById('searchBar').addEventListener('input', display);
document.getElementById('filterOptions').addEventListener('change', display);

display(); // Run on startup
</script>
